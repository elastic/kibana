version: "1"
# Bottom-Up strategy — one `document` KI per record in a bounded corpus.
#
# This is a working workflow, not a sketch. Edit the `consts` block and it runs as it stands;
# everything below `consts` is already correct and does not need to be rewritten to make it work.
#
# To adapt it:
#   1. Set every value in `consts`. `corpus_filter` is the agreed bound on what gets indicated —
#      it is a whole ES|QL line, so it can carry several conditions, and setting it to an empty
#      string takes the whole index.
#   2. Add any further fields the summary should draw on to the `KEEP` in `discover_documents`,
#      and name each new column in `rows`, which turns the positional ES|QL row into an object
#      the loop reads by field name.
#   3. Tell the prompt what these documents are. "Support article" and "incident record" produce
#      different summaries from the same text, and the model cannot tell which it is holding.
#
# One KI per document is the right granularity only for a bounded corpus of documents that each
# answer something on their own. When a pass produces many near-identically-titled KIs, the corpus
# wants a coarser strategy — see `analyze-and-improve`.
name: Document KI automation
description: >
  Summarises each document in a bounded corpus into its own Knowledge Indicator, carrying verified
  ES|QL access patterns that reach the live record.
enabled: true

consts:
  # The Context Engine AI index the indicators are written into.
  ai_index_id: my-ai-index
  # The index the documents are read from.
  source_index: my-source-index
  # The field used as the document's title.
  title_field: title
  # The field holding the document body.
  body_field: body
  # The agreed corpus filter, as a complete ES|QL line. Set to "" to take the whole index.
  corpus_filter: "| WHERE published_at >= NOW() - 365 days"
  # Upper bound on documents summarised in one run. Each costs a model call.
  max_documents: 50
  # Bodies are truncated before they reach the prompt so one long document cannot exhaust the
  # context window and silently drop the rest of the batch.
  body_max_chars: 12000

tags:
  - document-ki

triggers:
  - type: manual

steps:
  # Fetched once, outside the loop, and handed to every summary. Without it the model only ever
  # sees one document's text, so the access patterns it writes name fields that do not exist and
  # the KI fails verification — which costs the whole per-document model call. `LIMIT 0` returns
  # the flat `columns` list, one `{ name, type }` per field ES|QL can see, at a fraction of the
  # size of the nested mapping; it is pasted into every per-document prompt, so size is tokens.
  - name: fetch_fields
    type: elasticsearch.esql.query
    with:
      query: "FROM {{ consts.source_index }} | LIMIT 0"

  - name: discover_documents
    type: elasticsearch.esql.query
    with:
      query: |
        FROM {{ consts.source_index }} METADATA _id
        {{ consts.corpus_filter }}
        | EVAL body = SUBSTRING({{ consts.body_field }}, 1, {{ consts.body_max_chars }})
        | KEEP _id, {{ consts.title_field }}, body
        | LIMIT {{ consts.max_documents }}

  # ES|QL rows are positional arrays in `KEEP` order: _id, title, body. Naming the columns here
  # once is what lets the loop read `foreach.item.doc_id` instead of `foreach.item[0]`.
  - name: rows
    type: data.map
    items: "${{ steps.discover_documents.output.values }}"
    with:
      fields:
        doc_id: "${{ item[0] }}"
        title: "${{ item[1] }}"
        body: "${{ item[2] }}"

  - name: loop_documents
    type: foreach
    foreach: "${{ steps.rows.output }}"
    # One document that fails to summarise should not cost the rest of the batch. The failed
    # iteration stays in the execution history as a failed step; the fallback names the document.
    iteration-on-failure:
      continue: true
      fallback:
        - name: log_iteration_failure
          type: console
          with:
            message: "Skipped document {{ foreach.item.doc_id }} after a step failed."
    steps:
      - name: summarize_document
        type: ai.prompt
        # The default model for every prompt step. Change it to another AI connector id if the
        # user asks for one; leave it in place otherwise.
        connector-id: .google-gemini-3.5-flash-chat_completion
        # Transient connector errors, rate limits and structured-output parse failures are
        # retried; a prompt that can never satisfy its schema still fails every attempt.
        on-failure:
          retry:
            max-attempts: 3
            delay: "5s"
            strategy: exponential
            jitter: true
        with:
          systemPrompt: |
            You summarise source documents for a retrieval index. Ground every statement in the
            document text you are given. Never add background knowledge, and never state that the
            document says something it does not. Every ES|QL query you write must be
            syntactically and semantically valid.
          prompt: |
            You are writing a Knowledge Indicator for one document from the
            {{ consts.source_index }} index, so that an agent can find it and know what it covers
            without reading it.

            Document id: {{ foreach.item.doc_id }}
            Title: {{ foreach.item.title }}

            Body:
            {{ foreach.item.body }}

            Fields of {{ consts.source_index }} with their types — the only fields that exist,
            and the only ones your ES|QL may name:
            {{ steps.fetch_fields.output.columns | json }}

            Produce a structured summary of this document, using only its text above.

            This KI sits alongside one per other document in the corpus, so `summary` has to say
            what *this* document settles that its neighbours do not. A summary that would fit any
            document in the corpus is worse than useless: it makes the whole set unrankable.

            For `access_patterns` give up to 2 ways to reach this material as live data. Return
            an empty array rather than an invented query when the document is reached only by its
            id or no ES|QL against it is meaningful; a KI that only orients is allowed, a query
            that answers nothing is not. Each entry has:
            - question_type: a short label for the kind of question.
            - esql_template: a parameterized ES|QL query using named `?param` placeholders.
            - esql_example: the same query with real values substituted and a small LIMIT.
            - returns: a short description of what the query returns.

            At least one pattern retrieves this exact document; the others reach the documents
            around it — same topic, same period, whatever the corpus makes sensible.

            ES|QL correctness rules — esql_template and esql_example are both executed by the
            verifier, so both must be valid and runnable:
            - Start with `FROM {{ consts.source_index }}`; use `|` between commands. Every field
              you name must appear in the field list above — a field inferred from the document's
              subject matter rather than read off the list is the most common way this fails.
              Reading `_id` requires `FROM {{ consts.source_index }} METADATA _id`.
            - Quote string literals with double quotes; do not quote numbers or booleans.
            - Date literals are strict ISO-8601, e.g. "2024-01-01T00:00:00.000Z"; for relative
              ranges prefer `NOW() - 30 days`.
            - Keep esql_example runnable and cheap: end it with a small `| LIMIT n`.
          schema:
            type: object
            properties:
              display_name:
                type: string
                description: "Short title for this KI, <= 10 words"
              summary:
                type: string
                description: "What this document settles that its neighbours do not"
              topics:
                type: array
                items:
                  type: string
              answers_questions:
                type: array
                items:
                  type: string
                description: "Questions this document answers on its own"
              when_to_use:
                type: string
              access_patterns:
                type: array
                items:
                  type: object
                  properties:
                    question_type:
                      type: string
                    esql_template:
                      type: string
                    esql_example:
                      type: string
                    returns:
                      type: string
                  required: ["question_type", "esql_template", "esql_example", "returns"]
            required:
              ["display_name", "summary", "topics", "answers_questions", "when_to_use", "access_patterns"]

      # Built once so verify_ki and create_ki see exactly the same document.
      - name: assemble_ki
        type: data.set
        with:
          ki:
            type: document
            title: "{{ steps.summarize_document.output.content.display_name }}"
            # Carries the source document's own title and topics, which is what separates this KI
            # from its neighbours when their summaries read alike.
            description: >-
              {{ foreach.item.title }} —
              {{ steps.summarize_document.output.content.summary }}
            content: |
              # {{ foreach.item.title }}

              Source: {{ consts.source_index }} document {{ foreach.item.doc_id }}

              ## Summary
              {{ steps.summarize_document.output.content.summary }}

              ## Topics
              {% for topic in steps.summarize_document.output.content.topics -%}
              - {{ topic }}
              {% endfor %}
              ## Answers questions
              {% for q in steps.summarize_document.output.content.answers_questions -%}
              - {{ q }}
              {% endfor %}
              ## When to use
              {{ steps.summarize_document.output.content.when_to_use }}

              ## Access patterns
              {% if steps.summarize_document.output.content.access_patterns.size > 0 -%}
              {% for ap in steps.summarize_document.output.content.access_patterns -%}
              - [{{ ap.question_type }}] {{ ap.returns }}
                Template: `{{ ap.esql_template }}`
                Example: `{{ ap.esql_example }}`
              {% endfor %}
              {%- else -%}
              None. Fetch the document from {{ consts.source_index }} by its id, {{ foreach.item.doc_id }}.
              {%- endif %}
            tags:
              - "{{ consts.source_index }}"
              - document
            # What the KI was derived from: the index and the one document it summarizes. Who
            # wrote it is stamped by `createKi` itself.
            references:
              - uri: "index://{{ consts.source_index }}"
                relation: derived_from
              - uri: "doc://{{ consts.source_index }}/{{ foreach.item.doc_id }}"
                relation: derived_from
            attributes:
              # Every ES|QL the KI carries goes here, which is what the verifiers check. `default: nil`
              # omits the attribute when there are no access patterns, so the verifiers skip the KI
              # instead of failing an empty list.
              esql: "${{ steps.summarize_document.output.content.access_patterns | map: 'esql_example' | default: nil }}"
              topics: "${{ steps.summarize_document.output.content.topics }}"

      - name: verify_ki
        type: context-engine.verifyKi
        with:
          verifiers:
            - esql-valid-syntax
            - esql-valid-runtime
          ki: "${{ steps.assemble_ki.output.ki }}"

      - name: check_verification
        type: if
        condition: "steps.verify_ki.output.passed : true"
        steps:
          # `ki_id` pairs the index with the source document id, so a re-run replaces that
          # document's KI on an index destination (a data stream appends a newer revision of it)
          # rather than writing a second KI, and two indices holding the same document id do not
          # collide.
          - name: create_ki
            type: context-engine.createKi
            with:
              ai_index_id: "{{ consts.ai_index_id }}"
              ki_id: "{{ consts.source_index }}-{{ foreach.item.doc_id }}"
              ki: "${{ steps.assemble_ki.output.ki }}"
        else:
          - name: log_verification_failure
            type: console
            with:
              message: |
                KI verification failed for document
                {{ foreach.item.doc_id }}.
                Results: {{ steps.verify_ki.output.results | json }}
